    <script>
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');

        let width = canvas.width = window.innerWidth;
        let height = canvas.height = window.innerHeight;

        window.addEventListener('resize', () => {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
            blackHole.x = width / 2;
            blackHole.y = height / 2;
        });

        const blackHole = {
            x: width / 2,
            y: height / 2,
            mass: 1800,
            radius: 22,
            targetX: width / 2,
            targetY: height / 2
        };

        let isMouseDown = false;

        window.addEventListener('mousemove', (e) => {
            blackHole.targetX = e.clientX;
            blackHole.targetY = e.clientY;
        });

        window.addEventListener('mousedown', () => isMouseDown = true);
        window.addEventListener('mouseup', () => isMouseDown = false);

        class Particle {
            constructor() {
                this.reset(true);
            }

            reset(initial = false) {
                const angle = Math.random() * Math.PI * 2;
                const maxDist = Math.max(width, height) * 0.6;
                const dist = initial ? Math.random() * maxDist + 60 : maxDist;

                this.x = blackHole.x + Math.cos(angle) * dist;
                this.y = blackHole.y + Math.sin(angle) * dist;

                // Vận tốc quỹ đạo xoắn
                const orbitSpeed = Math.sqrt(blackHole.mass / dist) * 0.45;
                this.vx = -Math.sin(angle) * orbitSpeed + (Math.random() - 0.5);
                this.vy = Math.cos(angle) * orbitSpeed + (Math.random() - 0.5);

                this.radius = Math.random() * 1.6 + 0.6;
                this.hue = 200;
            }

            update() {
                const dx = blackHole.x - this.x;
                const dy = blackHole.y - this.y;
                const distSq = dx * dx + dy * dy;
                const dist = Math.sqrt(distSq);

                // Nuốt hạt khi chạm chân trời sự kiện (Event Horizon)
                if (dist < blackHole.radius) {
                    this.reset();
                    return;
                }

                const currentMass = isMouseDown ? blackHole.mass * 2.8 : blackHole.mass;
                const force = currentMass / (distSq + 120);

                // Lực hút tâm
                this.vx += (dx / dist) * force;
                this.vy += (dy / dist) * force;

                // Lực xoáy quỹ đạo
                this.vx += (-dy / dist) * force * 0.65;
                this.vy += (dx / dist) * force * 0.65;

                // Giảm ma sát nhẹ
                this.vx *= 0.988;
                this.vy *= 0.988;

                this.x += this.vx;
                this.y += this.vy;

                // Đổi màu hạt khi vào gần tâm (xanh lơ -> tím -> cam sáng)
                const proximity = 1 - Math.min(dist / (width * 0.4), 1);
                this.hue = 190 + proximity * 150;
            }

            draw() {
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fillStyle = `hsl(${this.hue}, 100%, 70%)`;
                ctx.fill();
            }
        }

        const particleCount = 2200;
        const particles = Array.from({ length: particleCount }, () => new Particle());

        function animate() {
            // Dời vị trí hố đen mượt mà theo con trỏ chuột
            blackHole.x += (blackHole.targetX - blackHole.x) * 0.06;
            blackHole.y += (blackHole.targetY - blackHole.y) * 0.06;

            // Tạo vệt mờ (Trail effect)
            ctx.fillStyle = 'rgba(5, 5, 13, 0.18)';
            ctx.fillRect(0, 0, width, height);

            // Vẽ đĩa bồi tụ phát sáng xung quanh hố đen (Accretion Disk Glow)
            const glow = ctx.createRadialGradient(
                blackHole.x, blackHole.y, 0,
                blackHole.x, blackHole.y, blackHole.radius * 3.5
            );
            glow.addColorStop(0, 'rgba(0, 0, 0, 1)');
            glow.addColorStop(0.35, 'rgba(20, 0, 40, 0.8)');
            glow.addColorStop(0.7, 'rgba(140, 60, 255, 0.25)');
            glow.addColorStop(1, 'transparent');

            ctx.fillStyle = glow;
            ctx.beginPath();
            ctx.arc(blackHole.x, blackHole.y, blackHole.radius * 3.5, 0, Math.PI * 2);
            ctx.fill();

            // Vẽ tâm hố đen hoàn toàn tối
            ctx.fillStyle = '#000000';
            ctx.beginPath();
            ctx.arc(blackHole.x, blackHole.y, blackHole.radius, 0, Math.PI * 2);
            ctx.fill();

            ctx.strokeStyle = 'rgba(180, 120, 255, 0.9)';
            ctx.lineWidth = 1.5;
            ctx.stroke();

            // Cập nhật và vẽ tất cả các hạt
            particles.forEach(p => {
                p.update();
                p.draw();
            });

            requestAnimationFrame(animate);
        }

        animate();
    </script>
